> **For agentic workers:** REQUIRED SUB-SKILL: 用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 按任务逐个实现。步骤用 `- [ ]` 复选框追踪。

**Goal:** 把 0128 从知网专用爬虫演进为「CNKI + scansci 双源、硬安全闸、统一输出」的采集层，CC 经 SSH 在校内备用机（校园 IP）上驱动。

**Architecture:** 一个轻量 `acq/` 包：纯逻辑内核（models/store/guard/dedup/net）+ 两个源 adapter（cnki 重构自现有代码、scansci 薄封装）+ 薄 CLI。agent 即编排，不自建重型框架。所有下载必过 RateGuard；校内源强制直连。

**Tech Stack:** Python 3.13（conda env）、Playwright(sync)、httpx、ddddocr、scansci-pdf、pytest。

## Global Constraints（每个任务都隐含遵守，数值照抄 spec）

- **校内源强制直连**：httpx 用 `trust_env=False` 且不传 proxy；Playwright 不传 proxy。CC 带代理驱动时也不串味。
- **硬安全闸**：所有 search/download 必过 `RateGuard`，agent 不可绕过。
- **输出契约**：`metadata.csv` 含 `source`、`dedup_key` 两列；全文落 `papers/<safe_title>.<ext>`；下载前 `Content-Type` 校验为 PDF。
- **不静默吞错**：禁止 `except: pass`，至少 `log`。
- **scansci 安全配置**：`scihub_enabled=false`、`batch_workers<=2`。
- **验证码**：本地 ddddocr，不上视觉模型。
- **编码**：全程 UTF-8；代码注释用中文。

> 🔴 **执行环境说明**：Phase 0–1（纯逻辑）可在**主力机**直接开发+测试。Phase 2–3 的**解析逻辑**可在主力机用 HTML 夹具测；但 **CNKI 实跑 / 验证码 / scansci 机构访问**必须在**校内备用机（校园 IP）**上验证 —— 标 `[需备用机]` 的步骤等 SSH 通了再做。

```python
# acq/models.py
"""源无关数据模型。"""
from dataclasses import dataclass, field
import re

def _norm_title(s: str) -> str:
    # 去标点/空白、转小写，做去重用的规范化标题
    return re.sub(r"[\s\W_]+", "", (s or "").strip().lower())

@dataclass
class PaperMeta:
    source: str
    source_id: str
    title: str
    authors: str = ""
    abstract: str = ""
    keywords: str = ""
    journal: str = ""
    pub_date: str = ""
    doi: str = ""
    institution: str = ""
    fund: str = ""
    cite_count: str = ""
    url: str = ""
    file_path: str = ""
    file_type: str = ""
    extra: dict = field(default_factory=dict)

    @property
    def dedup_key(self) -> str:
        if self.doi.strip():
            return self.doi.strip().lower()
        return _norm_title(self.title)

@dataclass
class DownloadResult:
    ok: bool
    path: str = ""
    file_type: str = ""
    reason: str = ""

@dataclass
class Query:
    terms: str
    mode: str = "keyword"   # keyword | professional
    date_from: str = ""
    date_to: str = ""
    raw: dict = field(default_factory=dict)  # 源名 -> 原生检索式（CNKI 专业检索透传）
```

```python
# acq/dedup.py
"""跨源去重：DOI 优先，否则规范化标题。"""
from dataclasses import fields
from acq.models import PaperMeta

def dedupe(metas: list[PaperMeta]) -> list[PaperMeta]:
    seen: dict[str, PaperMeta] = {}
    for m in metas:
        k = m.dedup_key
        if not k:
            seen[id(m)] = m  # 无 key 的不合并
            continue
        if k not in seen:
            seen[k] = m
        else:
            # 用新条目补已有条目的空字段
            old = seen[k]
            for f in fields(PaperMeta):
                if f.name == "extra":
                    continue
                if not getattr(old, f.name) and getattr(m, f.name):
                    setattr(old, f.name, getattr(m, f.name))
    return list(seen.values())
```

```python
# acq/guard.py
"""硬安全闸：限速 + 每日配额(持久化) + 熔断。所有下载必过。"""
import json
import time
import random
import logging
from pathlib import Path

log = logging.getLogger("acq.guard")

class RateGuard:
    def __init__(self, state_dir, config: dict, today: str):
        # today: 北京日期字符串 "YYYY-MM-DD"，由调用方传入（便于测试 & 跨天清零）
        self.dir = Path(state_dir); self.dir.mkdir(parents=True, exist_ok=True)
        self.cfg = config
        self.today = today
        self.f = self.dir / "guard_state.json"
        self.state = self._load()
        self._last_req: dict[str, float] = {}

    def _load(self) -> dict:
        if self.f.exists():
            try:
                return json.loads(self.f.read_text(encoding="utf-8"))
            except Exception as e:
                log.warning("guard 状态读失败，重置: %s", e)
        return {"date": self.today, "count": {}, "tripped": {}}

    def _save(self):
        tmp = self.f.with_suffix(".tmp")
        tmp.write_text(json.dumps(self.state, ensure_ascii=False), encoding="utf-8")
        tmp.replace(self.f)

    def _roll_day(self):
        if self.state.get("date") != self.today:
            self.state = {"date": self.today, "count": {}, "tripped": {}}
            self._save()

    def wait(self, source: str):
        self._roll_day()
        iv = self.cfg.get(source, {}).get("min_interval", 0)
        if iv:
            last = self._last_req.get(source, 0)
            delay = iv + random.uniform(0, iv * 0.5) - (time.time() - last)
            if delay > 0:
                log.info("[guard] %s 限速等待 %.1fs", source, delay)
                time.sleep(delay)
        self._last_req[source] = time.time()

    def can_download(self, source: str) -> bool:
        self._roll_day()
        if self.tripped(source):
            return False
        return self.remaining(source) > 0

    def remaining(self, source: str) -> int:
        self._roll_day()
        cap = self.cfg.get(source, {}).get("max_per_day", 10_000)
        used = self.state["count"].get(source, 0)
        return max(0, cap - used)

    def record(self, source: str):
        self._roll_day()
        self.state["count"][source] = self.state["count"].get(source, 0) + 1
        self._save()

    def trip(self, source: str, reason: str):
        log.error("[guard] 熔断 %s：%s", source, reason)
        self.state.setdefault("tripped", {})[source] = reason
        self._save()

    def tripped(self, source: str) -> bool:
        return source in self.state.get("tripped", {})
```

```python
# acq/store.py
"""统一元数据落盘（原子增量）+ 全文 manifest。"""
import csv, json, re
from pathlib import Path
from acq.models import PaperMeta

FIELDS = ["source","dedup_key","title","authors","abstract","keywords","journal",
          "pub_date","doi","institution","fund","cite_count","url","file_path","file_type"]

def _safe(name: str) -> str:
    return re.sub(r'[\\/*?:"<>|\n\r\t]', "_", name).strip()[:100]

class MetadataStore:
    def __init__(self, out_dir):
        self.dir = Path(out_dir); self.dir.mkdir(parents=True, exist_ok=True)
        self.papers = self.dir / "papers"; self.papers.mkdir(exist_ok=True)
        self.csv = self.dir / "metadata.csv"
        self.manifest = self.dir / "manifest.json"

    def _man(self) -> dict:
        return json.loads(self.manifest.read_text(encoding="utf-8")) if self.manifest.exists() else {}

    def is_downloaded(self, dedup_key: str) -> bool:
        return dedup_key in self._man()

    def mark_downloaded(self, dedup_key: str, path: str):
        m = self._man(); m[dedup_key] = path
        tmp = self.manifest.with_suffix(".tmp")
        tmp.write_text(json.dumps(m, ensure_ascii=False), encoding="utf-8"); tmp.replace(self.manifest)

    def paper_path(self, meta: PaperMeta, ext: str) -> Path:
        return self.papers / f"{_safe(meta.title)}.{ext}"

    def save_meta(self, metas: list[PaperMeta]):
        rows = {}
        if self.csv.exists():
            with open(self.csv, encoding="utf-8-sig", newline="") as f:
                for r in csv.DictReader(f):
                    rows[r.get("dedup_key","")] = r
        for m in metas:
            d = {k: getattr(m, k, "") for k in FIELDS if k != "dedup_key"}
            d["dedup_key"] = m.dedup_key
            if m.dedup_key in rows:
                for k in FIELDS:  # 补空字段
                    if d.get(k) and not rows[m.dedup_key].get(k):
                        rows[m.dedup_key][k] = d[k]
            else:
                rows[m.dedup_key] = d
        tmp = self.csv.with_suffix(".tmp")
        with open(tmp, "w", encoding="utf-8-sig", newline="") as f:
            w = csv.DictWriter(f, fieldnames=FIELDS, extrasaction="ignore")
            w.writeheader(); w.writerows(rows.values())
        tmp.replace(self.csv)
```
